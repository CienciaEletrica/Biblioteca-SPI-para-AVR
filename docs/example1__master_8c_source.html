1/**
2 * @file example1_master.c
3 * @author Tiago Henrique dos Santos - Ciência Elétrica
4 * @brief Exemplo SPI Master: Controle de fluxo e telemetria via UART.
5 *
6 * @details
7 * **Objetivos do Teste:**
8 * 1. Validar o Baud Rate de 125 kHz (Fosc/128).
9 * 2. Demonstrar o envio assíncrono (non-blocking) de comandos via interrupção.
10 * 3. Validar a recepção Full-Duplex (receber resposta do Slave enquanto envia).
11 *
12 * **Resultado Esperado:**
13 * - O terminal serial deve exibir "Enviando: XX | Recebido: YY".
14 * - O primeiro byte recebido será tipicamente 0x00 (inicialização).
15 * - Os bytes subsequentes refletirão a resposta processada pelo Slave no ciclo anterior.
16 *
17 * @example example1_master.c
18 */
19#define F_CPU 16000000UL
20#include <avr/io.h>
21#include <util/delay.h>
22#include <avr/interrupt.h>
23#include "spi.h"
24
25/**
26 * @brief Flag de erro para sinalizar overflow detectado na ISR.
27 */
28volatile bool flag_overflow_isr = false; // Global para teste
29
30/* Protótipos das funções auxiliares de UART */
31void uart_init(void);
32void uart_puts(const char *s);
33void uart_puthex8(uint8_t val);
34
35/**
36 * @brief Ponto de entrada do exemplo Master.
37 */
38int main(void) {
39 uart_init();
40 uart_puts("--- SPI MASTER: 125kHz Mode ---\r\n");
41
42 /**
43 * @note Configuração SPI Master:
44 * - Modo Assíncrono habilitado (use_interrupt = true).
45 * - Clock de 125 kHz (ideal para longas distâncias ou periféricos lentos).
46 */
47 spi_config_t master_cfg = {
48 .mode = SPI_MODE_MASTER,
49 .clock_div = SPI_CLOCK_DIV_128, // 125 kHz @ 16MHz
50 .cpol = 0, // Mode 0
51 .cpha = 0,
52 .bit_order = SPI_ORDER_MSB_FIRST,
53 .use_interrupt = true, // Motor assíncrono habilitado
54 .rx_callback = 0, // Leitura manual via buffer RX
55 .pins = {
56 .port = &PORTB,
57 .mosi_pin = PB3,
58 .miso_pin = PB4,
59 .sck_pin = PB5,
60 .ss_pin = PB2
61 }
62 };
63
64 /* Inicializa o driver */
65 if (!spi_init(&master_cfg)) {
66 uart_puts("Erro na inicializacao SPI\r\n");
67 while(1);
68 }
69
70 /* Registra o escravo 0 no pino SS (PB2) */
71 spi_register_slave(0, &PORTB, PB2);
72
73 /* Habilita interrupções globais */
74 sei();
75
76 uint8_t tx_data = 0xA5;
77 uint8_t rx_data = 0;
78
79 while (1) {
80 uart_puts("Enviando: "); uart_puthex8(tx_data);
81
82 /* Seleciona o dispositivo escravo (CS em nível baixo) */
83 spi_select_device(0); // Seleciona o device
84
85 /* Envio assíncrono: O byte vai para o buffer e a ISR gerencia a transmissão */
86 spi_write_async(tx_data);
87
88 /* Aguarda a conclusão da transmissão antes de ler o buffer RX */
89 while (spi_is_busy());
90
91 /* Verifica se houve resposta do Slave no buffer de recepção */
92 if (spi_rx_read(&rx_data)) {
93 uart_puts(" | Recebido do Slave: ");
94 uart_puthex8(rx_data);
95 uart_puts("\r\n");
96 }
97
98 tx_data++;
99 _delay_ms(500); // Delay para facilitar a leitura no monitor serial
100 }
101}

107/** @brief Inicializa a UART0 a 9600 bps. */
108void uart_init(void)
109{
110 uint16_t ubrr = 103;// Cálculo do UBRR para 9600 bps, modo normal - UBRR = (F_CPU / (16 * BAUD)) - 1
111 UBRR0H = (uint8_t)(ubrr >> 8);
112 UBRR0L = (uint8_t)ubrr;
113 UCSR0B = (1 << RXEN0) | (1 << TXEN0) | (1 << RXCIE0); // Habilita transmissãoo, recepção e interrupção Rx
114 UCSR0C = (1 << UCSZ01) | (1 << UCSZ00); // Formato do frame: 8 bits, 1 stop bit, sem paridade
115}
116
117/** @brief Envia um caractere via UART. */
118void uart_putc(char c)
119{
120 while (!(UCSR0A & (1 << UDRE0))); // espera buffer vazio
121 UDR0 = c;
122}
123
124/** @brief Envia uma string via UART. */
125void uart_puts(const char *s)
126{
127 while (*s) uart_putc(*s++);
128}
129
130/** @brief Envia um valor de 8 bits em formato hexadecimal. */
131void uart_puthex8(uint8_t val)
132{
133 const char hex[] = "0123456789ABCDEF";
134 uart_putc(hex[(val >> 4) & 0x0F]);
135 uart_putc(hex[val & 0x0F]);
136}
