1/**
2 * @file example1_slave.c
3 * @author Tiago Henrique dos Santos - Ciência Elétrica
4 * @brief Exemplo SPI Slave: Resposta dinâmica baseada em processamento.
5 *
6 * @details
7 * **Funcionamento:**
8 * 1. O Slave aguarda o sinal de clock do Master.
9 * 2. Ao receber um byte, a função @ref slave_callback é executada.
10 * 3. O Slave prepara o dado para a *próxima* transmissão usando @ref spi_slave_preload.
11 *
12 * **Nota sobre Full-Duplex:**
13 * No protocolo SPI, o dado que o Slave carrega no registrador agora só será
14 * deslocado para o Master na transferência seguinte. Por isso, usamos o callback
15 * para manter o pipeline de dados sempre atualizado.
16 *
17 * @example example1_slave.c
18 */
19
20#include "spi.h"
21#include <avr/io.h>
22#include <avr/interrupt.h>
23
24/**
25 * @brief Registrador de status simulado.
26 */
27volatile uint8_t status_reg = 0x10;
28
29/**
30 * @brief Callback de recepção para o modo Slave.
31 * @details Esta função é executada dentro da ISR logo após o término de um byte.
32 *
33 * @param[in] received_byte Byte que o Master acabou de enviar.
34 * @param[in] overflow Indica se o buffer interno falhou em armazenar o dado.
35 *
36 * @note No modo Slave, usamos o callback para carregar a resposta que o Master
37 * lerá no próximo ciclo de clock.
38 */
39void slave_callback(uint8_t received_byte, bool overflow) {
40 /* Exemplo: O Slave responde com o byte recebido incrementado de 1 */
41 spi_slave_preload(received_byte + 1);
42}
43
44/**
45 * @brief Ponto de entrada do exemplo Slave.
46 */
47int main(void) {
48 /**
49 * @note Configuração SPI Slave:
50 * - O pino SS deve ser configurado como entrada (feito automaticamente por @ref spi_init).
51 * - O clock_div é ignorado, pois o Slave segue o clock do Master.
52 */
53 spi_config_t slave_cfg = {
54 .mode = SPI_MODE_SLAVE,
55 .clock_div = SPI_CLOCK_DIV_4, // Ignorado no modo Slave (usa clock externo)
56 .cpol = 0, .cpha = 0,
57 .bit_order = SPI_ORDER_MSB_FIRST,
58 .use_interrupt = true,
59 .rx_callback = slave_callback,
60 .pins = {
61 .port = &PORTB,
62 .mosi_pin = PB3,
63 .miso_pin = PB4,
64 .sck_pin = PB5,
65 .ss_pin = PB2 // No Slave, SS deve ser entrada para o hardware SPI ativar
66 }
67 };
68
69 /* Inicializa o módulo SPI */
70 spi_init(&slave_cfg);
71
72 /**
73 * @brief Primeiro byte de resposta.
74 * @details Prepara o SPDR com um valor inicial (ex: 0x00) para que o Master
75 * não receba lixo eletrônico na primeiríssima leitura.
76 */
77 spi_slave_preload(0x00);
78
79 /* Habilita interrupções globais */
80 sei();
81
82 while(1) {
83 /**
84 * O Slave permanece em loop infinito. Toda a lógica de resposta
85 * é tratada de forma assíncrona pelo driver e pelo callback.
86 */
87 }
88}
