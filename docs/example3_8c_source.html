1/**
2 * @file example3.c
3 * @author Tiago Henrique dos Santos - Ciência Elétrica
4 * @brief Teste de Estresse: Validação de Overflow e Recuperação de Buffer.
5 *
6 * @details
7 * **Objetivos do Teste:**
8 * 1. Forçar o preenchimento total do buffer TX (32 bytes) para validar a trava de segurança.
9 * 2. Verificar se a flag `tx_overflow` é acionada corretamente ao tentar inserir o 33º byte.
10 * 3. Garantir que o sistema não sofra crash (estouro de pilha/RAM) ao escrever além da capacidade.
11 * 4. Validar o funcionamento das funções @ref spi_tx_available e @ref spi_tx_has_space.
12 *
13 * **Resultado Esperado:**
14 * - O terminal serial deve reportar exatamente 13 falhas (45 tentativas - 32 espaços).
15 * - A flag `tx_overflow` deve ser exibida como "SIM (Correto)".
16 * - O sistema deve processar os 32 bytes aceitos e retornar ao estado vazio (00 bytes) sem travar.
17 *
18 * @example example3.c
19 */
20#define F_CPU 16000000UL
21#include <avr/io.h>
22#include <util/delay.h>
23#include <avr/interrupt.h>
24#include "spi.h"
25
26/* Protótipos de UART para telemetria */
27void uart_init(void);
28void uart_puts(const char *s);
29void uart_puthex8(uint8_t val);
30void uart_putnewline(void);
31
32/**
33 * @brief Ponto de entrada do teste de estresse.
34 */
35int main(void) {
36 uart_init();
37 uart_puts("--- TESTE DE OVERFLOW SPI (Buffer: 32 bytes) ---\r\n");
38
39 /* 1. Configuração Master em velocidade reduzida para facilitar observação */
40 spi_config_t overflow_cfg = {
41 .mode = SPI_MODE_MASTER,
42 .clock_div = SPI_CLOCK_DIV_128,
43 .use_interrupt = true,
44 .rx_callback = NULL,
45 .pins = { .port = &PORTB, .mosi_pin = PB3, .miso_pin = PB4, .sck_pin = PB5, .ss_pin = PB2 }
46 };
47
48 if (!spi_init(&overflow_cfg)) {
49 uart_puts("Erro Init\r\n");
50 while(1);
51 }
52
53 spi_register_slave(0, &PORTB, PB2);
54 sei();
55
56 while (1) {
57 uart_puts("\r\nIniciando burst de 45 bytes...\r\n");
58
59 /* Limpa estados e buffers anteriores para o novo teste */
60 spi_tx_flush();
61 spi_rx_flush();
62
63 /**
64 * @brief Simulação de Carga Crítica.
65 * @note Bloqueamos a ISR momentaneamente (cli) para encher o buffer
66 * sem que o hardware SPI consuma os bytes imediatamente.
67 */
68 cli();
69
70 uint8_t falhas = 0;
71 for (uint8_t i = 0; i < 45; i++) {
72 /* Tenta enfileirar. A partir do 33º byte (índice 32), deve retornar false. */
73 if (!spi_write_async(0xAA)) {
74 falhas++;
75 }
76 }
77
78 /* Captura ocupação antes de liberar a transmissão */
79 uint8_t ocupado = spi_tx_available();
80
81 sei(); /* Libera a ISR para iniciar o esvaziamento do buffer via hardware */
82
83 /* --- Relatório de Diagnóstico via UART --- */
84 uart_puts("Tentativas que falharam: ");
85 uart_puthex8(falhas); // Esperado: 0D (13 decimal)
86 uart_putnewline();
87
88 uart_puts("Bytes aceitos no buffer: ");
89 uart_puthex8(ocupado); // Esperado: 20 (32 decimal)
90 uart_putnewline();
91
92 uart_puts("Flag tx_overflow: ");
93 uart_puts(spi_had_overflow() ? "SIM (Correto)" : "NAO (Erro)");
94 uart_putnewline();
95
96 /* Aguarda o esvaziamento total do buffer em background */
97 uart_puts("Transmitindo os dados aceitos...");
98 while (spi_is_busy());
99 uart_puts(" OK.\r\n");
100
101 uart_puts("Status final - Bytes no TX: ");
102 uart_puthex8(spi_tx_available());
103 uart_putnewline();
104
105 _delay_ms(5000); /* Aguarda 5 segundos para repetir o ciclo */
106 }
107}

112void uart_init(void)
113{
114 uint16_t ubrr = 103; // Cálculo do UBRR para 9600 bps, modo normal - UBRR = (F_CPU / (16 * BAUD)) - 1
115 UBRR0H = (uint8_t)(ubrr >> 8);
116 UBRR0L = (uint8_t)ubrr;
117 UCSR0B = (1 << RXEN0) | (1 << TXEN0) | (1 << RXCIE0); // Habilita transmissão, recepção e interrupção Rx
118 UCSR0C = (1 << UCSZ01) | (1 << UCSZ00); // Formato do frame: 8 bits, 1 stop bit, sem paridade
119}
120
121void uart_putc(char c)
122{
123 while (!(UCSR0A & (1 << UDRE0))); // espera buffer vazio
124 UDR0 = c;
125}
126
127void uart_puts(const char *s)
128{
129 while (*s) uart_putc(*s++);
130}
131
132void uart_puthex8(uint8_t val)
133{
134 const char hex[] = "0123456789ABCDEF";
135 uart_putc(hex[(val >> 4) & 0x0F]);
136 uart_putc(hex[val & 0x0F]);
137}
138
139void uart_putnewline(void)
140{
141 uart_putc('\r');
142 uart_putc('\n');
143}
