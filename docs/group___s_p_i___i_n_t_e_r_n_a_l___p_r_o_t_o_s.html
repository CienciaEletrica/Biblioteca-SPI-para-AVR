Definição na linha 49 do arquivo spi.c.

49 {
50 volatile uint8_t *ddr = cfg->pins.port - 1; // DDR é sempre PORT - 1 em AVR
51
52 if (cfg->mode == SPI_MODE_MASTER) {
53 // Master: SCK, MOSI e SS como saída. MISO como entrada.
54 *ddr |= (1 << cfg->pins.mosi_pin) | (1 << cfg->pins.sck_pin) | (1 << cfg->pins.ss_pin);
55 *ddr &= ~(1 << cfg->pins.miso_pin);
56 *(cfg->pins.port) |= (1 << cfg->pins.ss_pin); // Pull-up / High no SS
57 } else {
58 *ddr &= ~((1 << cfg->pins.mosi_pin) | (1 << cfg->pins.sck_pin) | (1 << cfg->pins.ss_pin));
59 *ddr |= (1 << cfg->pins.miso_pin);
60 *(cfg->pins.port) |= (1 << cfg->pins.ss_pin); // Pull-up no SS entrada
61 }
62}
