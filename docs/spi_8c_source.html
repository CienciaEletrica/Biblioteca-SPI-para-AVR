44/**
45 * @brief Configura a direção dos pinos físicos (DDR) baseada no modo Master/Slave.
46 * @param cfg Ponteiro para as configurações de pinos.
47 * @internal
48 */
49void spi_hw_init_pins(const spi_config_t *cfg) {
50 volatile uint8_t *ddr = cfg->pins.port - 1; // DDR é sempre PORT - 1 em AVR
51
52 if (cfg->mode == SPI_MODE_MASTER) {
53 // Master: SCK, MOSI e SS como saída. MISO como entrada.
54 *ddr |= (1 << cfg->pins.mosi_pin) | (1 << cfg->pins.sck_pin) | (1 << cfg->pins.ss_pin);
55 *ddr &= ~(1 << cfg->pins.miso_pin);
56 *(cfg->pins.port) |= (1 << cfg->pins.ss_pin); // Pull-up / High no SS
57 } else {
58 *ddr &= ~((1 << cfg->pins.mosi_pin) | (1 << cfg->pins.sck_pin) | (1 << cfg->pins.ss_pin));
59 *ddr |= (1 << cfg->pins.miso_pin);
60 *(cfg->pins.port) |= (1 << cfg->pins.ss_pin); // Pull-up no SS entrada
61 }
62}

64/**
65 * @brief Sincroniza as configurações da struct com os registradores SPCR e SPSR.
66 * @internal
67 */
68void spi_hw_apply_config(void) {
69 uint8_t spcr_val = (1 << SPE); // Habilita o hardware SPI
70
71 if (spi.config.mode == SPI_MODE_MASTER) spcr_val |= (1 << MSTR);
72 if (spi.config.bit_order == SPI_ORDER_LSB_FIRST) spcr_val |= (1 << DORD);
73 if (spi.config.cpol) spcr_val |= (1 << CPOL);
74 if (spi.config.cpha) spcr_val |= (1 << CPHA);
75
76 uint8_t spr = 0;
77 bool spi2x = false;
78
79 // Mapeamento do divisor de clock para os bits SPR0, SPR1 e SPI2X
80 switch (spi.config.clock_div) {
81 case SPI_CLOCK_DIV_2: spi2x = true; spr = 0; break;
82 case SPI_CLOCK_DIV_4: spr = 0; break;
83 case SPI_CLOCK_DIV_16: spr = 1; break;
84 case SPI_CLOCK_DIV_64: spr = 2; break;
85 case SPI_CLOCK_DIV_128: spr = 3; break;
86 }
87
88 spcr_val |= (spr << SPR0);
89 SPCR = spcr_val;
90
91 if (spi2x) SPSR |= (1 << SPI2X);
92 else SPSR &= ~(1 << SPI2X);
93
94 if (spi.config.use_interrupt) SPCR |= (1 << SPIE);
95}

97/**
98 * @brief Baixa o pino de Chip Select do escravo especificado.
99 * @param id ID do escravo (0 a SPI_MAX_SLAVES-1).
100 */
101void spi_hw_select_slave(uint8_t id) {
102 if (id >= SPI_MAX_SLAVES || spi.slaves[id].port == NULL) return;
103 *(spi.slaves[id].port) &= ~(spi.slaves[id].pin_mask); // CS = LOW
104 spi.current_slave_id = id;
105}
106
107/**
108 * @brief Sobe o pino de Chip Select do escravo atual.
109 */
110void spi_hw_deselect_slave(void) {
111 if (spi.current_slave_id >= SPI_MAX_SLAVES || spi.slaves[spi.current_slave_id].port == NULL) return;
112 *(spi.slaves[spi.current_slave_id].port) |= spi.slaves[spi.current_slave_id].pin_mask; // CS = HIGH
113 spi.current_slave_id = 255u;
114}

116/**
117 * @brief Inicia o pipeline de transmissão carregando o primeiro byte no SPDR.
118 * @return true se um byte foi disparado para o hardware.

120bool spi_hw_start_tx_pipeline(void) {
121 uint8_t byte_to_send;
122 if (!spi_tx_dequeue(&byte_to_send)) return false;
123 SPI_HW_WRITE_DATA(byte_to_send);
124 spi_state = SPI_STATE_TX_ACTIVE;
125 spi.busy = true;

129/**
130 * @brief Recupera o hardware de uma colisão de escrita (WCOL).

133void spi_hw_recover_collision(void) {
134 // Limpa flag WCOL lendo SPSR e SPDR
135 (void)SPSR;
136 (void)SPI_HW_READ_DATA();
137 // Pode tentar reiniciar pipeline ou apenas sinalizar erro
138 spi_state = SPI_STATE_ERROR_COLLISION;

144bool spi_init(const spi_config_t *cfg) {
145 if (!cfg) return false;
146 spi.config = *cfg;

149 spi_rx_flush();
150

152 spi_hw_apply_config();
153 return true;

156void spi_register_slave(uint8_t id, volatile uint8_t *port, uint8_t pin) {
157 if (id >= SPI_MAX_SLAVES) return;
158 spi.slaves[id].port = port;
159 spi.slaves[id].pin_mask = (1u << pin);
160 *port |= (1u << pin); // CS inicia em HIGH
161 *(port - 1) |= (1u << pin); // Configura DDR como saída
162}

172uint8_t spi_transfer_blocking(uint8_t data) {
173 if (spi.config.use_interrupt || spi.busy) return 0xFF;
174 uint16_t timeout = spi.timeout_default;
175
176 SPI_HW_WRITE_DATA(data);
177
178 // Polling da flag SPIF com proteção de timeout
179 while (!(SPSR & (1 << SPIF)) && timeout--);
180 return (timeout == 0) ? 0xFF : SPI_HW_READ_DATA();
181}

183bool spi_write_async(uint8_t data) {
184 if (!spi.config.use_interrupt) return false;
185
186 // Bloco atômico para evitar corrupção dos índices do buffer
187 ATOMIC_BLOCK(ATOMIC_RESTORESTATE) {
188 uint8_t head = spi.tx_head;
189 uint8_t tail = spi.tx_tail;
190 uint8_t used = (head >= tail) ? (head - tail) : (SPI_TX_BUFFER_SIZE - tail + head);
191
192 if (used >= (SPI_TX_BUFFER_SIZE - 1)) {
193 spi.tx_overflow = true;
194 return false;
195 }
196 spi.tx_buffer[spi.tx_head] = data;
197 spi.tx_head = spi_buf_next(spi.tx_head, SPI_TX_BUFFER_SIZE);
198 }
199
200 // Se o driver estiver parado, inicia a transmissão imediatamente
201 if (spi_state == SPI_STATE_IDLE) {
202 if (spi.current_slave_id < SPI_MAX_SLAVES) {
203 spi_hw_select_slave(spi.current_slave_id);
204 }
205 spi_hw_start_tx_pipeline();
206 }
207 return true;
208}

210bool spi_write_multi(const uint8_t *buf, uint8_t len, bool keep_cs_low) {
211 if (!spi.config.use_interrupt || spi.busy || len == 0) return false;
212
213 // Salva estado de CS antes de enfileirar
214 spi.multi_cs_active = keep_cs_low;
215
216 for (uint8_t i = 0; i < len; i++) {
217 if (!spi_write_async(buf[i])) {
218 // Se falhar no meio, opcionalmente limpamos tudo
219 spi_tx_flush();
220 return false;
221 }
222 }
223 return true;
224}

270/**
271 * @brief Altera dinamicamente o divisor de clock do SPI com segurança atômica.
272 *
273 * @details Esta função garante que a mudança de velocidade não ocorra durante
274 * uma transmissão ativa, o que causaria corrupção de dados ou
275 * comportamento errático no escravo.
276 *
277 * @param[in] new_div Novo divisor de clock baseado em @ref spi_clock_div_t.
278 */
279void spi_set_clock(spi_clock_div_t new_div) {
280 // 1. Segurança: Aguarda qualquer transmissão em background terminar.
281 while (spi_is_busy());
282
283 // 2. Cálculo dos bits conforme o enum definido em spi_types.h
284 uint8_t spr_bits = 0;
285 bool double_speed = false;
286
287 switch (new_div) {
288 case SPI_CLOCK_DIV_2: double_speed = true; spr_bits = 0; break;
289 case SPI_CLOCK_DIV_4: spr_bits = 0; break;
290 case SPI_CLOCK_DIV_16: spr_bits = 1; break;
291 case SPI_CLOCK_DIV_64: spr_bits = 2; break;
292 case SPI_CLOCK_DIV_128: spr_bits = 3; break;
293 }
294
295 // 3. SEÇÃO CRÍTICA: Aplicação atômica nos registradores de hardware
296 ATOMIC_BLOCK(ATOMIC_RESTORESTATE) {
297 // Atualiza a cópia na struct global para manter consistência
298 spi.config.clock_div = new_div;
299
300 // Limpa apenas os bits SPR1 e SPR0 (bits 1 e 0 do SPCR) e aplica os novos
301 SPCR = (SPCR & ~((1 << SPR1) | (1 << SPR0))) | (spr_bits & 0x03);
302
303 // Ajusta o bit SPI2X no SPSR (bit 0)
304 if (double_speed) {
305 SPSR |= (1 << SPI2X);
306 } else {
307 SPSR &= ~(1 << SPI2X);
308 }
309 }
310}

316/**
317 * @brief Interrupt Service Routine para o término de transferência SPI.
318 *
319 * @details Este é o motor da biblioteca no modo assíncrono. A cada byte
320 * concluído, a ISR:
321 * 1. Lê o byte recebido e o enfileira no buffer RX.
322 * 2. Executa o callback do usuário (se registrado).
323 * 3. Verifica se há mais bytes no buffer TX para enviar.
324 * 4. Gerencia o pino de Chip Select (CS) caso a transação termine.
325 * 5. Monitora e recupera colisões de escrita (WCOL).
326 */
327ISR(SPI_STC_vect) {
328 // 1. Captura imediata do dado recebido
329 uint8_t received = SPI_HW_READ_DATA();
330 bool rx_ok = spi_rx_enqueue(received);
331
332 // 2. Notificação da aplicação via callback
333 if (spi.config.rx_callback) spi.config.rx_callback(received, !rx_ok);
334
335 // 3. Gerenciamento do fluxo de transmissão (Pipeline)
336 uint8_t next_byte;
337 if (spi_tx_dequeue(&next_byte)) {
338 // Envia o próximo byte da fila
339 SPI_HW_WRITE_DATA(next_byte);
340 spi_state = SPI_STATE_TX_ACTIVE;
341 } else {
342 // Fila vazia: encerra a transação
343 spi_state = SPI_STATE_IDLE;
344 spi.busy = false;
345 // Desativa o escravo apenas se não for uma transação multi-byte mantida
346 if (!spi.multi_cs_active) spi_hw_deselect_slave();
347 }
348 // 4. Tratamento de erros de colisão de hardware
349 if (SPI_HW_WRITE_COLLISION_DETECTED()) spi_hw_recover_collision();
350}
