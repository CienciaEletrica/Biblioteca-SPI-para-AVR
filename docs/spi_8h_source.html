1/**
2 * @file spi.h
3 * @author Seu Nome / Equipe
4 * @brief Interface pública (API) da biblioteca SPI para microcontroladores AVR.
5 * @version 1.0
6 * @date 2024
7 *
8 * @details Esta biblioteca fornece uma camada de abstração (HAL) para o periférico SPI
9 * dos AVRs (foco ATmega328P). Suporta modos Master/Slave, operação síncrona
10 * (polling) e assíncrona (interrupção com buffers circulares).
11 *
12 * @note Esta é a camada de abstração que o usuário deve utilizar.
13 * Não inclua spi_internal.h nem acesse diretamente registradores do SPI.
14 *
15 * @warning Não chame funções SPI dentro de ISRs de outros periféricos sem cuidado
16 * (risco de deadlock ou corrupção de estado).
17 *
18 * @copyright Licença MIT (ou a sua preferência)
19 */
20

37/**
38 * @brief Inicializa o módulo SPI.
39 *
40 * Configura o hardware SPI de acordo com a estrutura fornecida. No ATmega328P,
41 * configura automaticamente PB5 (SCK), PB3 (MOSI), PB4 (MISO) e PB2 (SS).
42 *
43 * @param[in] cfg Ponteiro para a estrutura @ref spi_config_t com os parâmetros desejados.
44 * @return true se a inicialização foi bem-sucedida.
45 * @return false se houver parâmetros inválidos ou erro de hardware.
46 *
47 * @note Habilita pull-up no pino SS para evitar entrada acidental em modo slave.
48 */
49bool spi_init(const spi_config_t *cfg);
50

52/**
53 * @brief Altera dinamicamente o divisor de clock do SPI.
54 *
55 * @param[in] new_div Novo divisor de clock (baseado em @ref spi_clock_div_t).
56 * @note Esta função aguarda qualquer transmissão em andamento terminar antes de aplicar.
57 */
58void spi_set_clock(spi_clock_div_t new_div);

65 */
66
67/**
68 * @brief Registra um pino de Chip Select (CS) para um dispositivo escravo.
69 *
70 * @param[in] id Identificador do escravo (0 até @ref SPI_MAX_SLAVES - 1).
71 * @param[in] port Ponteiro para o registrador PORT (ex: &PORTB).
72 * @param[in] pin Número do pino no porto (0 a 7).
73 *
74 * @note O pino deve ser configurado como saída manualmente pelo usuário.
75 */
76void spi_register_slave(uint8_t id, volatile uint8_t *port, uint8_t pin);
77
78/**
79 * @brief Seleciona um dispositivo escravo (coloca CS em nível baixo).
80 *
81 * @param[in] id Identificador do escravo previamente registrado.
82 * @note Se houver um dispositivo selecionado anteriormente, ele será desativado primeiro.
83 */
84void spi_select_device(uint8_t id);

87/**
88 * @brief Desativa o dispositivo escravo atual (coloca CS em nível alto).
89 *
90 * @note Chamado automaticamente ao fim de transações multi-byte se `keep_cs_low` for false.

92void spi_deselect_device(void);

101/**
102 * @brief Transfere um byte de forma síncrona via polling.
103 *
104 * @param[in] data Byte a ser enviado.
105 * @return uint8_t Byte recebido durante a troca ou 0xFF em caso de timeout.
106 *
107 * @note Utiliza um contador simples para timeout definido em @ref spi_config_t.
108 */
109uint8_t spi_transfer_blocking(uint8_t data);

111/** @} */

113/**
114 * @defgroup SPI_ASYNC Transferência Assíncrona (Interrupt-driven)
115 * @{
116 */
117
118/**
119 * @brief Enfileira um byte para transmissão assíncrona.
120 *
121 * @param[in] data Byte a ser transmitido.
122 * @return true se o byte foi enfileirado no buffer TX.
123 * @return false se o buffer TX estiver cheio.
124 */
125bool spi_write_async(uint8_t data);
126
127/**
128 * @brief Inicia uma transação multi-byte assíncrona.
129 *
130 * @param[in] buf Ponteiro para o array de dados.
131 * @param[in] len Tamanho dos dados em bytes.
132 * @param[in] keep_cs_low Se true, mantém o pino CS baixo após o término.
133 *
134 * @return true se todos os bytes foram enfileirados com sucesso.
135 * @return false se não houver espaço suficiente no buffer ou módulo ocupado.
136 */
137bool spi_write_multi(const uint8_t *buf, uint8_t len, bool keep_cs_low);

139/**
140 * @brief Verifica o número de bytes disponíveis para leitura no buffer RX.

145/**
146 * @brief Lê um byte do buffer de recepção.
