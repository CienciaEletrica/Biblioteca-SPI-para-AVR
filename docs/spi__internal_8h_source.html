22/**
23 * @defgroup SPI_HW_MACROS Macros de Hardware
24 * @brief Acesso direto aos registradores SPCR, SPSR e SPDR.
25 * @{
26 */
27#define SPI_HW_ENABLE() do { SPCR |= (1 << SPE); } while (0)
28#define SPI_HW_DISABLE() do { SPCR &= ~(1 << SPE); } while (0)
29
30#define SPI_HW_SET_MASTER() do { SPCR |= (1 << MSTR); } while (0)
31#define SPI_HW_SET_SLAVE() do { SPCR &= ~(1 << MSTR); } while (0)
32
33#define SPI_HW_INTERRUPT_ENABLE() do { SPCR |= (1 << SPIE); } while (0)
34#define SPI_HW_INTERRUPT_DISABLE() do { SPCR &= ~(1 << SPIE); } while (0)
35
36/** @brief Verifica se a flag de interrupção (SPIF) está setada */
37#define SPI_HW_IS_INTERRUPT_FLAG_SET() (SPSR & (1 << SPIF))
38/** @brief Limpa a flag SPIF (requer leitura do SPSR seguida de leitura do SPDR) */
39#define SPI_HW_CLEAR_INTERRUPT_FLAG() do { (void)SPSR; (void)SPDR; } while (0)
40
41/** @brief Detecta colisão de escrita (WCOL) */
42#define SPI_HW_WRITE_COLLISION_DETECTED() (SPSR & (1 << WCOL))
43
44#define SPI_HW_DOUBLE_SPEED_ENABLE() do { SPSR |= (1 << SPI2X); } while (0)
45#define SPI_HW_DOUBLE_SPEED_DISABLE() do { SPSR &= ~(1 << SPI2X); } while (0)
46
47#define SPI_HW_WRITE_DATA(data) do { SPDR = (data); } while (0) /**< Escreve no registrador de dados (Inicia TX) */
48#define SPI_HW_READ_DATA() (SPDR) /**< Lê o registrador de dados recebidos */
49
50/** @brief Máscara para bits de clock SPR0 e SPR1 no registrador SPCR */
51#define SPI_HW_CLOCK_BITS(div) (((div) & 0x03) << SPR0)
52
53/**
54 * @brief Aplica divisor de clock e modo Double Speed.
55 * @param div Divisor (0..3).
56 * @param double_speed Booleano para modo 2x.
57 */
58#define SPI_HW_APPLY_CLOCK_DIV(div, double_speed) \
59 do { \
60 SPCR = (SPCR & ~((1<<SPR1)|(1<<SPR0))) | SPI_HW_CLOCK_BITS(div); \
61 if (double_speed) SPI_HW_DOUBLE_SPEED_ENABLE(); else SPI_HW_DOUBLE_SPEED_DISABLE(); \
62 } while (0)
63
64/** @brief Configura Polaridade (CPOL) e Fase (CPHA) */
65#define SPI_HW_SET_CPOL_CPHA(cpol, cpha) \
66 do { \
67 SPCR = (SPCR & ~((1<<CPOL)|(1<<CPHA))) | ((cpol)<<CPOL) | ((cpha)<<CPHA); \
68 } while (0)
69
70/** @brief Configura ordem dos bits (MSB ou LSB first) */
71#define SPI_HW_SET_DATA_ORDER(order) \
72 do { \
73 if (order == SPI_ORDER_LSB_FIRST) SPCR |= (1 << DORD); \
74 else SPCR &= ~(1 << DORD); \
75 } while (0)

98/**
99 * @brief Calcula o próximo índice do buffer circular.
100 * @note O tamanho do buffer deve ser obrigatoriamente potência de 2.
101 */
102static inline uint8_t spi_buf_next(uint8_t idx, uint8_t buf_size) {
103 return (uint8_t)((idx + 1u) & (uint8_t)(buf_size - 1u));
104}

106/**
107 * @brief Adiciona um byte à fila de transmissão.
108 * @return true se inserido, false se buffer cheio.
109 */
110static inline bool spi_tx_enqueue(uint8_t byte) {
111 uint8_t next_head = spi_buf_next(spi.tx_head, SPI_TX_BUFFER_SIZE);
112
113 if (next_head == spi.tx_tail) {
114 spi.tx_overflow = true;
115 return false;
116 }
117
118 spi.tx_buffer[spi.tx_head] = byte;
119 spi.tx_head = next_head; // escrita at�mica (8 bits)
120 return true;
121}
122
123/**
124 * @brief Remove um byte da fila de transmissão.
125 * @param[out] byte Ponteiro para armazenar o dado removido.
126 * @return true se havia dado para remover.
127 */
128static inline bool spi_tx_dequeue(uint8_t *byte) {
129 if (spi.tx_head == spi.tx_tail) {
130 return false;
131 }
132
133 *byte = spi.tx_buffer[spi.tx_tail];
134 spi.tx_tail = spi_buf_next(spi.tx_tail, SPI_TX_BUFFER_SIZE);
135 return true;
136}
137
138/**
139 * @brief Adiciona um byte recebido à fila RX (chamado na ISR).
140 */
141static inline bool spi_rx_enqueue(uint8_t byte) {
142 uint8_t next = spi_buf_next(spi.rx_head, SPI_RX_BUFFER_SIZE);
143
144 if (next == spi.rx_tail) {
145 spi.rx_overflow = true;
146 return false;
147 }
148
149 spi.rx_buffer[spi.rx_head] = byte;
150 spi.rx_head = next;
151 return true;
152}
153
154/**
155 * @brief Extrai um byte da fila RX para a aplicação.
156 */
157static inline bool spi_rx_dequeue(uint8_t *byte) {
158 if (spi.rx_head == spi.rx_tail) {
159 return false;
160 }
161
162 *byte = spi.rx_buffer[spi.rx_tail];
163 spi.rx_tail = spi_buf_next(spi.rx_tail, SPI_RX_BUFFER_SIZE);
164 return true;
165}
