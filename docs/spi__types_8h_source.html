1/**
2 * @file spi_types.h
3 * @brief Definições de tipos, enums e estruturas para a biblioteca SPI.
4 * @details Projetada para microcontroladores AVR (ATmega48/88/168/328).
5 * Foca em baixo consumo de recursos e segurança em ambientes com interrupção.
6 *
7 * @note Tamanhos de buffer devem ser potências de 2 para otimização de performance.
8 *
9 * @author Engenheiro Embarcado
10 * @date 2025-2026
11 */
12
13#ifndef SPI_TYPES_H
14#define SPI_TYPES_H
15
16#include <stdint.h>
17#include <stdbool.h>
18#include <avr/io.h>
19
20#ifndef NULL
21 /** @brief Definição padrão de NULL caso não esteja presente. */
22 #define NULL ((void *)0)
23#endif
24
25/**
26 * @defgroup SPI_CONFIG_CONSTS Constantes de Configuração
27 * @{
28 */
29
30/**
31 * @brief Tamanho do buffer de transmissão (TX).
32 * @warning Deve ser obrigatoriamente potência de 2.
33 */
34#define SPI_TX_BUFFER_SIZE 32u
35
36/**
37 * @brief Tamanho do buffer de recepção (RX).
38 * @warning Deve ser obrigatoriamente potência de 2.
39 */
40#define SPI_RX_BUFFER_SIZE 32u
41
42/* Verificação estática de potência de 2 */
43#if (SPI_TX_BUFFER_SIZE & (SPI_TX_BUFFER_SIZE - 1)) != 0
44 #error "SPI_TX_BUFFER_SIZE deve ser potência de 2"
45#endif
46#if (SPI_RX_BUFFER_SIZE & (SPI_RX_BUFFER_SIZE - 1)) != 0
47 #error "SPI_RX_BUFFER_SIZE deve ser potência de 2"
48#endif
49/** @brief Quantidade máxima de escravos gerenciáveis pelo driver. */
50#define SPI_MAX_SLAVES 4u
51
52/** @} */
53
54/**
55 * @brief Modos de operação do periférico SPI.
56 */
57typedef enum {
58 SPI_MODE_MASTER = 1, /**< Mestre: gera o sinal de clock e controla o CS. */
59 SPI_MODE_SLAVE = 0 /**< Escravo: reage ao sinal de clock externo. */
60} spi_mode_t;
61
62/**
63 * @brief Opções de prescaler para o clock SPI (Modo Master).
64 * @details Os valores mapeiam a combinação dos bits SPR1:0 e SPI2X.
65 */
66typedef enum {
67 SPI_CLOCK_DIV_4 = 0b000, /**< fosc / 4 */
68 SPI_CLOCK_DIV_16 = 0b001, /**< fosc / 16 */
69 SPI_CLOCK_DIV_64 = 0b010, /**< fosc / 64 */
70 SPI_CLOCK_DIV_128 = 0b011, /**< fosc / 128 */
71 SPI_CLOCK_DIV_2 = 0b100 /**< fosc / 2 (Modo Double Speed) */
72} spi_clock_div_t
73
74/**
75 * @brief Ordem de transmissão de dados no barramento.
76 */
77typedef enum {
78 SPI_ORDER_MSB_FIRST = 0, /**< Bit mais significativo enviado primeiro. */
79 SPI_ORDER_LSB_FIRST = 1 /**< Bit menos significativo enviado primeiro. */
80} spi_bit_order_t;
81
82/**
83 * @brief Códigos de status e erros das operações SPI.
84 */
85typedef enum {
86 SPI_OK = 0, /**< Operação concluída com sucesso. */
87 SPI_BUSY = -1, /**< Recurso ocupado ou transmissão em curso. */
88 SPI_TIMEOUT = -2, /**< Tempo limite de espera excedido. */
89 SPI_OVERFLOW = -3, /**< Buffer cheio, dados foram perdidos. */
90 SPI_NO_SLAVE = -4, /**< ID de escravo não registrado. */
91 SPI_INVALID_PARAM = -5 /**< Parâmetro de configuração inválido. */
92} spi_status_t;
93
94/**
95 * @brief Mapeamento de hardware para os pinos do barramento.
96 */
97typedef struct {
98 volatile uint8_t *port; /**< Endereço do registrador PORT (ex: &PORTB). */
99 uint8_t mosi_pin; /**< Número do pino MOSI. */
100 uint8_t miso_pin; /**< Número do pino MISO. */
101 uint8_t sck_pin; /**< Número do pino SCK. */
102 uint8_t ss_pin; /**< Número do pino SS (Hardware Slave Select). */
103} spi_pins_t;
104
105/**
106 * @brief Estrutura de configuração para inicialização do SPI.
107 */
108typedef struct {
109 spi_mode_t mode; /**< Modo Master ou Slave. */
110 spi_clock_div_t clock_div; /**< Divisor de clock (relevante apenas em Master). */
111 uint8_t cpol : 1; /**< Polaridade: 0=Repouso Baixo, 1=Repouso Alto. */
112 uint8_t cpha : 1; /**< Fase: 0=Amostra na subida, 1=Amostra na descida. */
113 spi_bit_order_t bit_order; /**< Ordem dos bits (MSB/LSB). */
114 bool use_interrupt; /**< true para modo assíncrono (ISR), false para polling. */
115
116 spi_pins_t pins; /**< Definição física dos pinos. */
117
118 /**
119 * @brief Função de callback para eventos de recepção.
120 * @param received_byte Byte que acabou de chegar.
121 * @param overflow_detected Indica se o buffer RX falhou em armazenar o byte.
122 */
123 void (*rx_callback)(uint8_t received_byte, bool overflow_detected);
124} spi_config_t;
125
126/**
127 * @brief Estrutura de controle (Handle) do driver SPI.
128 * @internal Armazena o estado em tempo de execução e buffers.
129 */
130typedef struct {
131 spi_config_t config; /**< Cópia da configuração ativa. */
132
133 volatile bool busy; /**< Indica operação em progresso. */
134 volatile bool multi_cs_active; /**< Mantém CS baixo entre transferências. */
135
136 uint8_t tx_buffer[SPI_TX_BUFFER_SIZE]; /**< Buffer circular de saída. */
137 uint8_t rx_buffer[SPI_RX_BUFFER_SIZE]; /**< Buffer circular de entrada. */
138
139 volatile uint8_t tx_head; /**< Índice de inserção no TX. */
140 volatile uint8_t tx_tail; /**< Índice de remoção no TX. */
141 volatile uint8_t rx_head; /**< Índice de inserção no RX. */
142 volatile uint8_t rx_tail; /**< Índice de remoção no RX. */
143
144 volatile bool tx_overflow; /**< Flag de erro de estouro no TX. */
145 volatile bool rx_overflow; /**< Flag de erro de estouro no RX. */
146
147 /** @brief Tabela de pinos CS para múltiplos escravos. */
148 struct {
149 volatile uint8_t *port; /**< Porta do pino CS. */
150 uint8_t pin_mask; /**< Máscara de bit do pino CS. */
151 } slaves[SPI_MAX_SLAVES];
152
153 uint8_t current_slave_id; /**< ID do escravo atualmente selecionado. */
154 uint16_t timeout_default; /**< Timeout padrão para operações bloqueantes. */
155} spi_handle_t;
156
157/**
158 * @brief Instância única do driver SPI.
159 */
160extern spi_handle_t spi;
161
162/**
163 * @name Macros de Status

166
167/** @brief Máscara para cálculo de índice circular. */
168#define SPI_BUFFER_MASK(size) ((size) - 1u)
169
170/** @brief Retorna a quantidade de bytes ocupados no buffer TX. */
171#define SPI_TX_AVAILABLE() ((uint8_t)(spi.tx_head - spi.tx_tail))
172
173/** @brief Retorna a quantidade de bytes prontos para leitura no buffer RX. */
174#define SPI_RX_AVAILABLE() ((uint8_t)(spi.rx_head - spi.rx_tail))
175
176/** @} */
177#endif /* SPI_TYPES_H */
